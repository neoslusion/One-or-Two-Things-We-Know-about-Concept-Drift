\section{Background: The MMD Drift Signal}
% #############################################################################

\begin{frame}[shrink]{MMD: The Unsupervised Drift Signal}
  \begin{columns}[c]
    \begin{column}{0.52\textwidth}
      \textbf{Why unsupervised?}
      \begin{itemize}
        \small
        \item Supervised detection needs ground-truth labels right after each prediction.
        \item In IIoT / streaming, labels are delayed or expensive $\to$ monitor the \textbf{raw distribution $P(X)$} instead.
      \end{itemize}
      \vspace{0.3em}
      \textbf{Maximum Mean Discrepancy:} map points into an RKHS and compare their mean embeddings (``centers of mass'').
      \begin{block}{\small MMD Statistic}
        \small
        $\widehat{\text{MMD}}^2 = \underbrace{\text{Within } P}_{\text{self-sim}} + \underbrace{\text{Within } Q}_{\text{self-sim}} - \underbrace{\text{Cross } P\leftrightarrow Q}_{\text{dis-sim}}$
      \end{block}
      {\small $\widehat{\text{MMD}}^2 \approx 0$: same distribution;\quad $\widehat{\text{MMD}}^2 \gg 0$: \textbf{drift candidate}.}
    \end{column}
    \begin{column}{0.48\textwidth}
      \centering
      \includegraphics[width=\linewidth,height=0.72\textheight,keepaspectratio]{image/mmd_variance_change.png}
    \end{column}
  \end{columns}
\end{frame}


\begin{frame}[shrink]{From MMD Trace to ShapeDD's Triangle}
  \begin{columns}[c]
    \begin{column}{0.5\textwidth}
      \textbf{Stream $\to$ trace:}
      \begin{itemize}
        \small
        \item A sliding window of size $2l_1$ splits into $W_{\text{ref}}$ and $W_{\text{test}}$; their MMD at each step forms a continuous trace $\sigma(t)$.
        \item Stable data $\to$ small noise floor; crossing a drift $\to$ $\sigma(t)$ rises into a peak.
      \end{itemize}
      \vspace{0.2em}
      \textbf{ShapeDD's insight --- the \emph{triangle}:}
      \begin{itemize}
        \small
        \item A sudden drift yields a \textbf{triangular} $\sigma(t)$ (linear contamination as the window crosses the drift) --- detect that \emph{shape}, not a raw threshold.
        \item Peak height $\sigma(t_c) =$ \textbf{drift magnitude} $\approx \text{MMD}^2(P,Q)$.
      \end{itemize}
      \begin{alertblock}{\small Bottleneck}
        \small
        Significance via a \textbf{permutation test} rebuilds the kernel thousands of times. We replace it with \textbf{IDW weighting} $+$ \textbf{Gamma-null} (next).
      \end{alertblock}
    \end{column}
    \begin{column}{0.48\textwidth}
      \centering
      \includegraphics[width=\linewidth,height=0.8\textheight,keepaspectratio]{image/shapedd_intuition.png}
    \end{column}
  \end{columns}
\end{frame}

\section{Layer 1: Detection}
% #############################################################################


\begin{frame}[shrink]{Inverse Density-Weighted MMD (IDW-MMD)}
  \begin{columns}[c]
    \begin{column}{0.5\textwidth}
      \textbf{Limitation of standard MMD:}
      \begin{itemize}
        \small
        \item Dense majority clusters dominate the statistic.
        \item Early drifts at distribution \textbf{boundaries} get washed out.
      \end{itemize}
      \vspace{0.3em}
      \begin{exampleblock}{\small IDW weighting (the idea)}
        \small
        Weight each point by \textbf{inverse local density}:
        $$w_i \propto \frac{1}{\sqrt{d_i}+\epsilon},\qquad d_i=\sum_{j} k(x_i,x_j).$$
        Dense core $\to$ small $w_i$;\; sparse \textbf{boundary} $\to$ large $w_i$.
      \end{exampleblock}
      {\footnotesize Weighted statistic: $\widehat{\mathrm{MMD}}^2_w=\sum_{i,j} w_i w_j\,k(x_i,x_j)+\sum_{i,j} w'_i w'_j\,k(y_i,y_j)-2\,\overline{k(x_i,y_j)}.$}
    \end{column}
    \begin{column}{0.48\textwidth}
      \centering
      \includegraphics[width=\linewidth,height=0.6\textheight,keepaspectratio]{theory_idw_weighting.png}\\[0.2em]
      {\scriptsize \textbf{Result:} sensitive to early boundary changes, no extra false alarms on the bulk.}
    \end{column}
  \end{columns}
\end{frame}


\begin{frame}{Statistical Calibration (Gamma-Null Approximation)}
	\begin{columns}[T]
		\begin{column}{0.5\textwidth}
			\textbf{The Permutation Bottleneck:}
			\begin{itemize}
				\small
				\item Standard ShapeDD runs $B=2500$ permutations to estimate p-value.
				\item Each permutation re-builds the kernel matrix $\to$ extremely slow.
			\end{itemize}
			
			\vspace{0.8em}
			\textbf{Gamma-Null Approximation Solution:}
			\begin{itemize}
				\small
				\item Under $H_0$, the MMD statistic is approximately Gamma-distributed.
				\item We estimate the first two moments (mean \& variance) from a small number of bootstrap shuffles ($B=20$), reusing the kernel matrix.
				\item Matches accuracy while achieving a \textbf{7--10$\times$ end-to-end speedup}.
			\end{itemize}
		\end{column}
		\begin{column}{0.48\textwidth}
			\centering
			\includegraphics[width=\linewidth,keepaspectratio]{../../results/plots/gamma_approximation.png}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{ShapeDD-IDW in Action: MMD Trace $\rightarrow$ Single Drift Event}
	\begin{columns}[c]
		\begin{column}{0.63\textwidth}
			\centering
			\includegraphics[width=\linewidth,height=0.84\textheight,keepaspectratio]{theory_detection_flow.png}
		\end{column}
		\begin{column}{0.35\textwidth}
			\textbf{The detector, end-to-end:}
			\begin{enumerate}
				\small
				\item \textbf{(a)} Stream $\to$ compute the \textbf{IDW-MMD trace} $\sigma(t)$.
				\item \textbf{(b)} \textbf{Isolate peaks:} smooth $\sigma(t)$, keep points above $\bar\sigma+\mathrm{std}$. A short \textbf{blip} stays sub-threshold $\to$ ignored.
				\item \textbf{(c)} \textbf{Validate \& group:} the Gamma-null $p$-value (Bonferroni-adjusted) confirms the peak; the whole super-threshold \textbf{burst collapses to one event} within tolerance $\delta$.
			\end{enumerate}
			\vspace{0.4em}
			\begin{exampleblock}{\small Outcome}
				\scriptsize
				One calibrated, label-free detection per true drift --- no permutation test, few false alarms.
			\end{exampleblock}
		\end{column}
	\end{columns}
\end{frame}

\section{Layer 2: Classification}
% #############################################################################

\begin{frame}[shrink]{CDT-MSW --- Supervised Classification Baseline}
	\begin{columns}[T]
		\begin{column}{0.55\textwidth}
			\textbf{CDT-MSW} (Guo et al., 2022) groups drift into:
			\begin{itemize}
				\item \textbf{TCD (Transient):} Sudden, Blip, Recurrent.
				\item \textbf{PCD (Progressive):} Gradual, Incremental.
			\end{itemize}

			\vspace{0.5em}
			\textbf{Growth Process:}
			\begin{itemize}
				\item Tracks the variance of accuracy degradation over a multi-sliding window.
				\item Short degradation $\to$ TCD; sustained degradation $\to$ PCD.
			\end{itemize}

			\vspace{0.5em}
			\begin{alertblock}{Limitation}
				CDT-MSW needs \textbf{ground-truth labels} at every step to compute the accuracy curve --- impractical for IIoT streams where labels arrive late or never.
			\end{alertblock}
		\end{column}
		\begin{column}{0.42\textwidth}
			\centering
			\includegraphics[width=\linewidth,height=0.65\textheight,keepaspectratio]{image/cdt_msw.png}

			\vspace{0.5em}
			\begin{exampleblock}{\small Motivation for SE-CDT}
				\small
				Replace the accuracy curve with the \emph{shape} of the MMD signal $\sigma(t)$. The same TCD/PCD distinction can then be made \textbf{without any labels at runtime}.
			\end{exampleblock}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{SE-CDT: Reading the MMD Shape (9 Attributes)}
  \textbf{Type drift without labels by reading the \emph{shape} of $\sigma(t)$ at the detection point:}
  \vspace{0.2em}
  \begin{columns}[c]
    \begin{column}{0.4\textwidth}
      \begin{block}{\footnotesize Geometric (peak structure)}
        \scriptsize
        $n_p$ (peak count), WR (width ratio), CV (coeff.\ of variation), SNR, PPR (peak proximity), DPAR (dual-peak amplitude).
      \end{block}
      \begin{block}{\footnotesize Temporal (time behaviour)}
        \scriptsize
        LTS (linear trend), SDS (step score), MS (monotonicity).
      \end{block}
      {\scriptsize \textbf{$+$ VR} --- one \emph{raw-data} signal (not from $\sigma(t)$); the slow-drift tie-breaker.}
    \end{column}
    \begin{column}{0.58\textwidth}
      \centering
      \includegraphics[width=\linewidth,height=0.74\textheight,keepaspectratio]{theory_peak_anatomy.png}
    \end{column}
  \end{columns}
\end{frame}


\begin{frame}{SE-CDT: Variance Ratio (VR) for Slow Drifts}
  \textbf{Challenge:} Gradual and Incremental both give wide, flat MMD humps --- shape alone can't separate them. \textbf{Idea:} read the \emph{raw-window} variance --- Gradual $=$ mixture of concepts $\Rightarrow$ variance \textbf{spike}; Incremental $=$ single shift $\Rightarrow$ \textbf{flat}.
  \vspace{0.1em}
  $$\mathrm{VR}=\frac{\max_i \operatorname{Var}\!\big(X_{[i,\,i+l_2)}\big)}{\operatorname{Var}\!\big(X_{[0,\,l_1)}\big)},\qquad \mathrm{VR}>1.3\Rightarrow\textbf{Gradual},\quad \mathrm{VR}<1.1\Rightarrow\textbf{Incremental}.$$
  \vspace{-0.3em}
  \begin{center}
    \includegraphics[width=0.92\linewidth,height=0.52\textheight,keepaspectratio]{theory_variance_ratio.png}
  \end{center}
\end{frame}

\begin{frame}{Hybrid Pipeline: IDW-MMD to Detect, Standard MMD to Classify}
	\vfill
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\textbf{Why use two MMD variants instead of one?}
			\begin{itemize}
				\scriptsize
				\item \textbf{IDW-MMD:} Very sharp, filters out noise and stabilizes Type-I error ($\alpha=0.05$) on stable windows. But its density weights flatten gradual transitions.
				\item \textbf{Standard MMD:} Slower but preserves natural geometry of MMD trace (rising trends, duration info) needed for classification.
			\end{itemize}
		\end{column}
		\begin{column}{0.48\textwidth}
			\begin{block}{Gated Execution Strategy}
				\scriptsize
				\begin{itemize}
					\item \textbf{L1 Detect:} Run cheap \textcolor{univblue}{IDW-MMD} continuously.
					\item \textbf{L2 Classify:} Compute \textcolor{univblue}{Standard MMD} only at detection point $t^*$ on the candidate window to extract shape.
				\end{itemize}
			\end{block}
		\end{column}
	\end{columns}

\section{Evaluation \& Results}
% #############################################################################

\begin{frame}{Datasets \& Experimental Setup}
	\vfill
	\begin{columns}[c]
		\begin{column}{0.53\textwidth}
			\textbf{14 Evaluation Datasets (Grouped):}
			\begin{itemize}
				\scriptsize
				\item \textbf{Sudden $P(X)$:} Gaussian Shift (Moderate), GCS (3 concepts), 4 $\times$ Random Uniform (Sensitivity tests: Mild to Ultra Severe).
				\item \textbf{Blip $P(X)$:} RBF Blips (short-term recurring centroid shifts).

\begin{frame}[noframenumbering,shrink]{Backup: SE-CDT Classification Algorithm}
	\textbf{Algorithm: SE-CDT --- Unsupervised Drift Type Classification}
	\vspace{0.5em}
	\begin{algorithmic}[1]
	    \scriptsize
		\REQUIRE Drift magnitude signal $\sigma(t)$ from Detection Module
		\REQUIRE Raw data window $W$ for Variance Ratio
		\ENSURE Drift type: \{TCD-Blip, TCD-Sudden, TCD-Recurrent, PCD-Gradual, PCD-Incremental\}
		\STATE $\sigma_s \leftarrow \text{GaussianFilter}(\sigma,\; \sigma_g)$
		\STATE $P \leftarrow \text{FindPeaks}(\sigma_s,\; \bar{\sigma}_s + \alpha\,\sigma_{\mathrm{std}})$
		\STATE Compute features: $n_p \leftarrow |P|$, WR, CV, SNR, PPR, DPAR, LTS, SDS, MS
		\STATE $\mathrm{VR} \leftarrow \text{VarianceRatio}(W)$
		\STATE $\ell \leftarrow \text{EstimateDriftDuration}(\sigma_s)$
		\IF{$n_p \leq \tau_{n}^{\mathrm{blip}}$ \textbf{and} $\mathrm{WR} < \tau_{\mathrm{WR}}^{\mathrm{blip}}$ \textbf{and} $\text{BlipProfile}(\dots)$}
			\RETURN \textbf{TCD-Blip}
		\ENDIF
		\IF{$\ell > \tau_{\ell}$} 
			\IF{$\text{IsIncremental}(\mathrm{VR}, \mathrm{LTS}, \dots)$}
				\RETURN \textbf{PCD-Incremental}
			\ELSE
				\RETURN \textbf{PCD-Gradual}
			\ENDIF
		\ENDIF
		\IF{$n_p \leq \tau_{n}^{\mathrm{sud}}$ \textbf{and} $\mathrm{WR} < \tau_{\mathrm{WR}}$ \textbf{and} $\mathrm{SNR} > \tau_{\mathrm{SNR}}$}
			\RETURN \textbf{TCD-Sudden}
		\ENDIF
		\IF{$n_p \geq \tau_{n}^{\mathrm{rec}}$ \textbf{and} $\mathrm{CV} < \tau_{\mathrm{CV}}$ \textbf{and} $\mathrm{LTS} < \tau_{\mathrm{LTS}}$}
			\RETURN \textbf{TCD-Recurrent}
		\ENDIF
		\IF{$\text{IsIncremental}(\mathrm{VR}, \mathrm{LTS}, \dots)$}
			\RETURN \textbf{PCD-Incremental}
		\ELSE
			\RETURN \textbf{PCD-Gradual}
		\ENDIF
	\end{algorithmic}
\end{frame}

\begin{frame}[noframenumbering,shrink]{Backup: IDW-MMD vs Optimally-Weighted MMD (Bharti et al.)}
	\textbf{Question:} ``How is IDW-MMD different from the Optimally-Weighted MMD of Bharti et al. (2023)?''

	\vspace{0.5em}
	\begin{columns}[T]
		\begin{column}{0.48\textwidth}
			\begin{block}{Bharti et al. (OW-MMD)}
				\begin{itemize}
					\item Domain: \textbf{likelihood-free inference} (ABC).
					\item Weights chosen by optimising against a target posterior.
					\item Requires solving a constrained convex optimisation per call.
					\item Not designed for streaming drift detection.
				\end{itemize}
			\end{block}
		\end{column}
		\begin{column}{0.48\textwidth}
			\begin{exampleblock}{IDW-MMD (this work)}
				\begin{itemize}
					\item Domain: \textbf{drift detection} in fast data streams.
					\item Weights from a closed-form heuristic: $w_i \propto 1/(\sqrt{d_i} + \epsilon)$.
					\item Cost: $O(n^2)$ for the kernel matrix, $O(n)$ extra for the weights --- no optimisation step.
					\item Same \textit{weighted MMD} family, different objective and very different runtime.
				\end{itemize}
			\end{exampleblock}
		\end{column}
	\end{columns}

	\vspace{0.5em}
	\textbf{Summary:} both are weighted MMD variants, but they target different problems (ABC vs streaming drift), use different weight functions, and have very different runtime profiles.
\end{frame}

	\textbf{Question:} ``Recurrent drift requires concept matching --- how does Concept Memory provide it?''

	\vspace{0.5em}
	\begin{columns}[T]
		\begin{column}{0.52\textwidth}
			\textbf{The detection challenge:}
			\begin{itemize}
				\item A Recurrent stream is a sequence of \textit{repeated} concept switches; each switch produces an MMD peak that, in isolation, looks like a Sudden drift.
				\item Without memory of past concepts, repeated visits are scored independently $\to$ Recurrent collapses into the Sudden branch.
			\end{itemize}

			\vspace{0.5em}
			\textbf{How Concept Memory resolves it:}
			\begin{enumerate}
				\item During every \textit{stable period}, take a snapshot of the current data window and store it in a ring buffer of size $K = 8$.
				\item When a new drift fires, compute Standard MMD between the new post-drift window and every snapshot.
				\item If the smallest MMD $< \mu = 0.15$ $\to$ override the rule output to \textbf{Recurrent}.
			\end{enumerate}
		\end{column}
		\begin{column}{0.45\textwidth}
			\begin{exampleblock}{Why $\mu = 0.15$?}
				\small
				$\mu = 0.05$ falls inside the MMD noise floor (false matches between independent stationary windows). An empirical sweep on H0 calibration data shows $\mu = 0.15$ separates ``revisited concept'' from ``new concept'' reliably.
			\end{exampleblock}

			\vspace{0.3em}
			\begin{alertblock}{Remaining limitations}
				\small
				Two failure modes remain on the harder Recurrent variants:
				\begin{itemize}
					\item Recurrent events that are \textit{too far apart} push the original snapshot out of the ring buffer.
					\item Repeated Gradual streams saturate after a few events, so the post-drift window matches a stored snapshot and is labelled Recurrent rather than Gradual.
				\end{itemize}
				$\to$ Future work: learned threshold and longer memory.
			\end{alertblock}
		\end{column}
	\end{columns}
\end{frame}

% --- Backup 5: Why not DL ---
\begin{frame}[noframenumbering,shrink]{Backup: ``Why not deep learning?''}
	\textbf{Question:} ``Couldn't a deep learning model solve this more accurately?''

	\vspace{0.5em}
	\begin{columns}[T]
		\begin{column}{0.48\textwidth}
			\begin{alertblock}{Deep models for drift detection --- limitations}
				\begin{itemize}
					\item Need a large labelled dataset to train --- the very thing we don't have on a streaming IIoT pipeline.
					\item Higher inference latency per window than a kernel computation.
					\item No closed-form theoretical bound on the test statistic.
					\item Decision is hard to explain to a domain expert.
				\end{itemize}
			\end{alertblock}
		\end{column}
		\begin{column}{0.48\textwidth}
			\begin{exampleblock}{Why MMD / ShapeDD fits this problem}
				\begin{itemize}
					\item Fully unsupervised --- no training run before deployment.
					\item Theoretical guarantees: Triangle Shape Theorem, Type-I error from H0 calibration.
					\item Decision rule is human-readable (a small tree of geometric features).
					\item One mechanism works across $P(X)$ shifts of arbitrary type.
				\end{itemize}
			\end{exampleblock}

			\vspace{0.3em}
			\textbf{Future direction:} keep the kernel framework but plug in a learned feature extractor for high-dimensional inputs (Chapter 6).
		\end{column}
	\end{columns}
\end{frame}

% --- Backup 7: Permutation vs Gamma null ---
\begin{frame}[noframenumbering,shrink]{Backup: Permutation test vs Gamma-null p-value}
	\textbf{Question:} ``Removing the permutation test --- is the Gamma-null p-value still trustworthy?''

	\vspace{0.5em}
	\begin{columns}[T]
		\begin{column}{0.48\textwidth}
			\begin{alertblock}{Permutation test (original ShapeDD)}
				\begin{itemize}
					\item Reshuffles indices many times per window.
					\item Empirical null, no distributional assumption.
					\item Each shuffle re-builds the kernel matrix $\to$ heavy compute.
				\end{itemize}
			\end{alertblock}
		\end{column}
		\begin{column}{0.48\textwidth}
			\begin{exampleblock}{Gamma-null p-value (IDW-MMD)}
				\begin{itemize}
					\item Moment-matched Gamma fit to the null MMD distribution (Gretton et al.\ 2009).
					\item Estimates the first two moments from a small number of index shuffles, reusing the kernel matrix.
					\item Complexity $O(n^2)$ for the kernel plus a few cheap shuffles.
				\end{itemize}
			\end{exampleblock}
		\end{column}
	\end{columns}

	\vspace{0.5em}
	\textbf{Verification:} The detection $F_1$ remains comparable to the permutation baseline; the H0-calibration on three stationary distributions reports Standard MMD and IDW-MMD inside the 95\% CI of $\alpha = 0.05$ ($0.025$--$0.075$), while the SE-CDT composite is anti-conservative at $0.080$--$0.140$ (up to $\approx 2.8\times\,\alpha$), reflecting peak-selection and sliding-window dependence rather than the per-candidate test itself. End-to-end runtime is slightly faster than the ShapeDD baseline.
\end{frame}

% --- Backup 12: Why the slow subtypes are low ---
\begin{frame}[noframenumbering,shrink]{Backup: Why are Gradual (27.2\%) and Incremental (30.3\%) low?}
	\textbf{Question:} ``The two slow types are the weak point --- what exactly goes wrong?''

	\vspace{0.5em}
	\begin{columns}[T]
		\begin{column}{0.5\textwidth}
			\textbf{Two systematic confusions:}
			\begin{enumerate}
				\small
				\item \textbf{Gradual $\leftrightarrow$ Incremental overlap:} Both land in ambiguous Variance Ratio bands ($\sim$47\% of Incremental read as Gradual).
				\item \textbf{Saturation $\to$ Recurrent:} In repeated slow drifts, $P(X)$ stops changing. The post-drift window matches stored snapshots, labelling it Recurrent ($\sim$57\% of Gradual events).
			\end{enumerate}
		\end{column}
		\begin{column}{0.48\textwidth}
			\begin{exampleblock}{\small Why CAT is unaffected}
				\small
				Both confusions stay \emph{inside} PCD, preserving TCD/PCD accuracy (CAT $=80.1\%$). Confusing Gradual with Incremental is low-cost for adaptation.
			\end{exampleblock}

			\vspace{0.3em}
			\begin{alertblock}{\small Fix (future work)}
				\scriptsize
				Replace fixed VR thresholds with learned classifiers and separate Concept-Memory match step from shape classification.
			\end{alertblock}
		\end{column}
	\end{columns}
\end{frame}
